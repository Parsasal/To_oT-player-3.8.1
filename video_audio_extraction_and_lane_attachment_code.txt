/**
 * High-performance streaming audio extractor and DSP processor.
 * Capable of processing files from a few seconds up to 6 hours without high memory usage (OOM-safe).
 */
object VideoAudioExtractor {

    private const val TAG = "VideoAudioExtractor"
    const val MAX_DURATION_MS = 6 * 60 * 60 * 1000L // 6 hours in milliseconds

    data class VideoInfo(
        val uri: Uri,
        val fileName: String,
        val durationMs: Long,
        val formattedDuration: String,
        val isDurationValid: Boolean,
        val artist: String?,
        val title: String?
    )

    /**
     * Inspects the selected video URI to retrieve duration and metadata safely.
     * Uses ParcelFileDescriptor when available to prevent memory issues with large files.
     */
    fun inspectVideo(context: Context, videoUri: Uri): VideoInfo {
        val retriever = MediaMetadataRetriever()
        var durationMs = 0L
        var title: String? = null
        var artist: String? = null
        var displayName = "video_audio_${System.currentTimeMillis()}"
        var pfd: ParcelFileDescriptor? = null

        try {
            pfd = try {
                context.contentResolver.openFileDescriptor(videoUri, "r")
            } catch (_: Throwable) {
                null
            }

            if (pfd != null) {
                retriever.setDataSource(pfd.fileDescriptor)
            } else {
                retriever.setDataSource(context, videoUri)
            }

            val durationStr = retriever.extractMetadata(MediaMetadataRetriever.METADATA_KEY_DURATION)
            durationMs = durationStr?.toLongOrNull() ?: 0L
            title = retriever.extractMetadata(MediaMetadataRetriever.METADATA_KEY_TITLE)
            artist = retriever.extractMetadata(MediaMetadataRetriever.METADATA_KEY_ARTIST)

            // Try to resolve display name from content resolver
            try {
                context.contentResolver.query(videoUri, null, null, null, null)?.use { cursor ->
                    val nameIndex = cursor.getColumnIndex(android.provider.OpenableColumns.DISPLAY_NAME)
                    if (nameIndex != -1 && cursor.moveToFirst()) {
                        val name = cursor.getString(nameIndex)
                        if (!name.isNullOrBlank()) {
                            displayName = name.substringBeforeLast(".")
                        }
                    }
                }
            } catch (t: Throwable) {
                Log.w(TAG, "Could not query display name: ${t.message}")
            }
        } catch (t: Throwable) {
            Log.e(TAG, "Error inspecting video: ${t.message}", t)
        } finally {
            try { pfd?.close() } catch (_: Throwable) {}
            try { retriever.release() } catch (_: Throwable) {}
        }

        val formattedDuration = formatDuration(durationMs)
        val isDurationValid = durationMs in 1..MAX_DURATION_MS

        return VideoInfo(
            uri = videoUri,
            fileName = displayName,
            durationMs = durationMs,
            formattedDuration = formattedDuration,
            isDurationValid = isDurationValid,
            artist = artist,
            title = title ?: displayName
        )
    }

    /**
     * Backward-compatible simple extraction
     */
    suspend fun extractAudioToMp3(
        context: Context,
        videoUri: Uri,
        outputFileName: String,
        onProgress: (Float) -> Unit
    ): Result<File> = withContext(Dispatchers.IO) {
        val settings = AudioEffectsSettings(mode = AudioExtractionMode.FULL)
        val result = extractAudioWithEffects(context, videoUri, outputFileName, settings, onProgress)
        result.map { files -> files.first() }
    }

    /**
     * Extracts audio from video via streaming chunk pipeline, applies DSP effects (EQ, reverb, widening, limiter),
     * and encodes to standard crystal-clear WAV audio without loading the full file into memory.
     * Guaranteed safe for 55-minute or multi-hour videos!
     */
    suspend fun extractAudioWithEffects(
        context: Context,
        videoUri: Uri,
        outputBaseName: String,
        settings: AudioEffectsSettings,
        onProgress: (Float) -> Unit
    ): Result<List<File>> = withContext(Dispatchers.IO) {
        val extractor = MediaExtractor()
        var retriever: MediaMetadataRetriever? = null
        var decoder: MediaCodec? = null
        var pfd: ParcelFileDescriptor? = null

        var writer1: StreamingWavWriter? = null
        var writer2: StreamingWavWriter? = null
        val generatedFiles = mutableListOf<File>()

        try {
            val contentResolver = context.contentResolver
            pfd = contentResolver.openFileDescriptor(videoUri, "r")
                ?: return@withContext Result.failure(Exception("Unable to open video file."))

            extractor.setDataSource(pfd.fileDescriptor)

            retriever = MediaMetadataRetriever()
            retriever.setDataSource(pfd.fileDescriptor)
            val durationStr = retriever.extractMetadata(MediaMetadataRetriever.METADATA_KEY_DURATION)
            val durationUs = (durationStr?.toLongOrNull() ?: 1000L) * 1000L
            val totalDurationMs = durationUs / 1000L

            if (durationUs > MAX_DURATION_MS * 1000L) {
                return@withContext Result.failure(
                    Exception("Video duration exceeds the maximum allowed limit (6 hours).")
                )
            }

            // Find audio track
            var audioTrackIndex = -1
            var audioFormat: MediaFormat? = null
            for (i in 0 until extractor.trackCount) {
                val format = extractor.getTrackFormat(i)
                val mime = format.getString(MediaFormat.KEY_MIME) ?: ""
                if (mime.startsWith("audio/")) {
                    audioTrackIndex = i
                    audioFormat = format
                    break
                }
            }

            if (audioTrackIndex == -1 || audioFormat == null) {
                return@withContext Result.failure(Exception("No valid audio track found in this video clip."))
            }

            extractor.selectTrack(audioTrackIndex)

            val mimeType = audioFormat.getString(MediaFormat.KEY_MIME) ?: "audio/mp4a-latm"
            var sampleRate = if (audioFormat.containsKey(MediaFormat.KEY_SAMPLE_RATE)) {
                audioFormat.getInteger(MediaFormat.KEY_SAMPLE_RATE)
            } else 44100
            var channelCount = if (audioFormat.containsKey(MediaFormat.KEY_CHANNEL_COUNT)) {
                audioFormat.getInteger(MediaFormat.KEY_CHANNEL_COUNT)
            } else 2
            var pcmEncoding = if (audioFormat.containsKey(MediaFormat.KEY_PCM_ENCODING)) {
                audioFormat.getInteger(MediaFormat.KEY_PCM_ENCODING)
            } else AudioFormat.ENCODING_PCM_16BIT

            try {
                audioFormat.setInteger(MediaFormat.KEY_PCM_ENCODING, AudioFormat.ENCODING_PCM_16BIT)
            } catch (_: Throwable) {}

            // Prepare output dir
            val outputDir = File(context.filesDir, "extracted_audio").apply { mkdirs() }
            val cleanBase = outputBaseName
                .replace(Regex("[^a-zA-Z0-9._\\u0600-\\u06FF -]"), "_")
                .trim()
                .removeSuffix(".mp3")
                .removeSuffix(".wav")
                .removeSuffix(".m4a")
                .ifBlank { "extracted_audio_${System.currentTimeMillis()}" }

            val totalEstimatedSamples = (totalDurationMs * sampleRate * 2) / 1000L
            val audioProcessor = StreamingAudioProcessor(settings, sampleRate, 2, totalEstimatedSamples)

            val startUs = (settings.startMs.coerceAtLeast(0L)) * 1000L
            val endUs = if (settings.endMs > settings.startMs) settings.endMs * 1000L else durationUs
            val splitUs = (if (settings.splitMs in 1000 until totalDurationMs) settings.splitMs else totalDurationMs / 2) * 1000L

            when (settings.mode) {
                AudioExtractionMode.FULL -> {
                    val outFile = File(outputDir, "$cleanBase.wav")
                    writer1 = StreamingWavWriter(outFile, sampleRate, 2)
                    generatedFiles.add(outFile)
                }
                AudioExtractionMode.TRIM -> {
                    val outFile = File(outputDir, "${cleanBase}_trimmed.wav")
                    writer1 = StreamingWavWriter(outFile, sampleRate, 2)
                    generatedFiles.add(outFile)
                    if (startUs > 0) {
                        try {
                            extractor.seekTo(startUs, MediaExtractor.SEEK_TO_PREVIOUS_SYNC)
                        } catch (_: Throwable) {}
                    }
                }
                AudioExtractionMode.SPLIT -> {
                    val outFile1 = File(outputDir, "${cleanBase}_Part1.wav")
                    val outFile2 = File(outputDir, "${cleanBase}_Part2.wav")
                    writer1 = StreamingWavWriter(outFile1, sampleRate, 2)
                    writer2 = StreamingWavWriter(outFile2, sampleRate, 2)
                    generatedFiles.add(outFile1)
                    generatedFiles.add(outFile2)
                }
            }

            decoder = MediaCodec.createDecoderByType(mimeType)
            decoder.configure(audioFormat, null, null, 0)
            decoder.start()

            val bufferInfo = MediaCodec.BufferInfo()
            var sawInputEOS = false
            var sawOutputEOS = false
            val kTimeOutUs = 10000L
            var currentPresentationTimeUs = 0L

            while (!sawOutputEOS) {
                if (!sawInputEOS) {
                    val inputBufIndex = decoder.dequeueInputBuffer(kTimeOutUs)
                    if (inputBufIndex >= 0) {
                        val inputBuf = decoder.getInputBuffer(inputBufIndex)
                        if (inputBuf != null) {
                            inputBuf.clear()
                            val sampleSize = extractor.readSampleData(inputBuf, 0)
                            if (sampleSize < 0) {
                                decoder.queueInputBuffer(inputBufIndex, 0, 0, 0L, MediaCodec.BUFFER_FLAG_END_OF_STREAM)
                                sawInputEOS = true
                            } else {
                                val presentationTimeUs = extractor.sampleTime
                                decoder.queueInputBuffer(inputBufIndex, 0, sampleSize, presentationTimeUs, 0)
                                extractor.advance()
                            }
                        }
                    }
                }

                val outputBufferIndex = decoder.dequeueOutputBuffer(bufferInfo, kTimeOutUs)
                if (outputBufferIndex >= 0) {
                    if (bufferInfo.size > 0) {
                        currentPresentationTimeUs = bufferInfo.presentationTimeUs
                        val outputBuf = decoder.getOutputBuffer(outputBufferIndex)
                        if (outputBuf != null) {
                            outputBuf.position(bufferInfo.offset)
                            outputBuf.limit(bufferInfo.offset + bufferInfo.size)
                            val chunk = ByteArray(bufferInfo.size)
                            outputBuf.get(chunk)

                            // 1. Convert float PCM to standard 16-bit PCM if decoder produced 32-bit float
                            val rawPcm16 = if (pcmEncoding == AudioFormat.ENCODING_PCM_FLOAT) {
                                convertFloatToPcm16(chunk)
                            } else {
                                chunk
                            }

                            // 2. Normalize channels to standard stereo (2-channel)
                            // Multi-channel (5.1/7.1) is downmixed cleanly preserving singer's vocals
                            // Mono is duplicated to L & R so headphones play equally
                            val stereoChunk = when {
                                channelCount == 1 -> convertMonoToStereo(rawPcm16)
                                channelCount == 2 -> rawPcm16
                                channelCount > 2 -> downmixMultiChannelToStereo(rawPcm16, channelCount)
                                else -> rawPcm16
                            }

                            // Process chunk through streaming DSP effects pipeline (if any effect is active)
                            val processedChunk = if (audioProcessor.hasEffects) {
                                audioProcessor.processChunk(stereoChunk, 0, stereoChunk.size)
                            } else {
                                stereoChunk
                            }

                            when (settings.mode) {
                                AudioExtractionMode.FULL -> {
                                    writer1?.write(processedChunk)
                                }
                                AudioExtractionMode.TRIM -> {
                                    if (currentPresentationTimeUs in startUs..endUs) {
                                        writer1?.write(processedChunk)
                                    } else if (currentPresentationTimeUs > endUs && sawInputEOS) {
                                        sawOutputEOS = true
                                    }
                                }
                                AudioExtractionMode.SPLIT -> {
                                    if (currentPresentationTimeUs < splitUs) {
                                        writer1?.write(processedChunk)
                                    } else {
                                        writer2?.write(processedChunk)
                                    }
                                }
                            }
                        }
                    }

                    decoder.releaseOutputBuffer(outputBufferIndex, false)

                    if (durationUs > 0) {
                        val p = (currentPresentationTimeUs.toFloat() / durationUs.toFloat()).coerceIn(0f, 1f)
                        onProgress(p)
                    }

                    if ((bufferInfo.flags and MediaCodec.BUFFER_FLAG_END_OF_STREAM) != 0) {
                        sawOutputEOS = true
                    }
                } else if (outputBufferIndex == MediaCodec.INFO_OUTPUT_FORMAT_CHANGED) {
                    val newFormat = decoder.outputFormat
                    if (newFormat.containsKey(MediaFormat.KEY_SAMPLE_RATE)) {
                        sampleRate = newFormat.getInteger(MediaFormat.KEY_SAMPLE_RATE)
                    }
                    if (newFormat.containsKey(MediaFormat.KEY_CHANNEL_COUNT)) {
                        channelCount = newFormat.getInteger(MediaFormat.KEY_CHANNEL_COUNT)
                    }
                    if (newFormat.containsKey(MediaFormat.KEY_PCM_ENCODING)) {
                        pcmEncoding = newFormat.getInteger(MediaFormat.KEY_PCM_ENCODING)
                    }
                    writer1?.updateFormat(sampleRate, 2)
                    writer2?.updateFormat(sampleRate, 2)
                }
            }

            // Flush and finalize WAV headers
            try { writer1?.close() } catch (_: Throwable) {}
            try { writer2?.close() } catch (_: Throwable) {}

            onProgress(1.0f)
            Result.success(generatedFiles)
        } catch (t: Throwable) {
            Log.e(TAG, "Audio extraction failed: ${t.message}", t)
            Result.failure(Exception(t.message ?: "Audio extraction failed"))
        } finally {
            try { writer1?.close() } catch (_: Throwable) {}
            try { writer2?.close() } catch (_: Throwable) {}
            try { decoder?.stop() } catch (_: Throwable) {}
            try { decoder?.release() } catch (_: Throwable) {}
            try { retriever?.release() } catch (_: Throwable) {}
            try { extractor.release() } catch (_: Throwable) {}
            try { pfd?.close() } catch (_: Throwable) {}
        }
    }

    private fun convertFloatToPcm16(floatBytes: ByteArray): ByteArray {
        val numFloats = floatBytes.size / 4
        if (numFloats <= 0) return ByteArray(0)
        val out = ByteArray(numFloats * 2)
        val inBuf = ByteBuffer.wrap(floatBytes).order(ByteOrder.LITTLE_ENDIAN)
        val outBuf = ByteBuffer.wrap(out).order(ByteOrder.LITTLE_ENDIAN)
        for (i in 0 until numFloats) {
            val f = inBuf.float
            val s = (f * 32767.0f).coerceIn(-32768.0f, 32767.0f).toInt().toShort()
            outBuf.putShort(s)
        }
        return out
    }

    private fun convertMonoToStereo(monoPcm16: ByteArray): ByteArray {
        val numSamples = monoPcm16.size / 2
        if (numSamples <= 0) return ByteArray(0)
        val out = ByteArray(numSamples * 4)
        val inBuf = ByteBuffer.wrap(monoPcm16).order(ByteOrder.LITTLE_ENDIAN)
        val outBuf = ByteBuffer.wrap(out).order(ByteOrder.LITTLE_ENDIAN)
        for (i in 0 until numSamples) {
            val s = inBuf.short
            outBuf.putShort(s)
            outBuf.putShort(s)
        }
        return out
    }

    private fun downmixMultiChannelToStereo(pcm16Bytes: ByteArray, channels: Int): ByteArray {
        if (channels <= 2) return pcm16Bytes
        val bytesPerSample = 2
        val frameSize = channels * bytesPerSample
        val numFrames = pcm16Bytes.size / frameSize
        if (numFrames <= 0) return ByteArray(0)
        val out = ByteArray(numFrames * 4)
        val inBuf = ByteBuffer.wrap(pcm16Bytes).order(ByteOrder.LITTLE_ENDIAN)
        val outBuf = ByteBuffer.wrap(out).order(ByteOrder.LITTLE_ENDIAN)

        val centerWeight = 0.707f
        val surroundWeight = 0.707f

        for (f in 0 until numFrames) {
            val fl = inBuf.short.toFloat()
            val fr = inBuf.short.toFloat()
            val c = if (channels > 2) inBuf.short.toFloat() else 0f
            val lfe = if (channels > 3) inBuf.short.toFloat() else 0f
            val sl = if (channels > 4) inBuf.short.toFloat() else 0f
            val sr = if (channels > 5) inBuf.short.toFloat() else 0f
            for (ch in 6 until channels) {
                inBuf.short
            }

            val left = (fl + c * centerWeight + sl * surroundWeight).coerceIn(-32768f, 32767f).toInt().toShort()
            val right = (fr + c * centerWeight + sr * surroundWeight).coerceIn(-32768f, 32767f).toInt().toShort()

            outBuf.putShort(left)
            outBuf.putShort(right)
        }
        return out
    }

    /**
     * Streaming WAV container writer that writes PCM data in real-time and rewrites the standard 44-byte RIFF header on close.
     */
    private class StreamingWavWriter(
        private val outputFile: File,
        private var sampleRate: Int,
        private var channelCount: Int = 2
    ) : AutoCloseable {
        private val raf = RandomAccessFile(outputFile, "rw")
        private var totalDataBytes = 0L
        private var isClosed = false

        init {
            raf.setLength(0)
            // 44 dummy bytes reserved for header
            raf.write(ByteArray(44))
        }

        fun updateFormat(newSampleRate: Int, newChannelCount: Int = 2) {
            if (newSampleRate > 0) sampleRate = newSampleRate
            if (newChannelCount > 0) channelCount = newChannelCount
        }

        fun write(bytes: ByteArray) {
            if (bytes.isEmpty() || isClosed) return
            raf.write(bytes)
            totalDataBytes += bytes.size
        }

        override fun close() {
            if (isClosed) return
            isClosed = true
            try {
                val numChannels = channelCount.coerceIn(1, 2)
                val bytesPerSample = 2 // 16-bit PCM
                val blockAlign = numChannels * bytesPerSample
                val byteRate = sampleRate * blockAlign
                val subchunk2Size = totalDataBytes.coerceAtMost(Int.MAX_VALUE.toLong()).toInt()
                val chunkSize = 36 + subchunk2Size

                val header = ByteBuffer.allocate(44).apply {
                    order(ByteOrder.LITTLE_ENDIAN)
                    put("RIFF".toByteArray())
                    putInt(chunkSize)
                    put("WAVE".toByteArray())
                    put("fmt ".toByteArray())
                    putInt(16) // Subchunk1Size (16 for PCM)
                    putShort(1.toShort()) // AudioFormat (1 = PCM uncompressed)
                    putShort(numChannels.toShort())
                    putInt(sampleRate)
                    putInt(byteRate)
                    putShort(blockAlign.toShort())
                    putShort((bytesPerSample * 8).toShort())
                    put("data".toByteArray())
                    putInt(subchunk2Size)
                }

                raf.seek(0)
                raf.write(header.array())
            } catch (t: Throwable) {
                Log.e(TAG, "Error writing WAV header: ${t.message}")
            } finally {
                raf.close()
            }
        }
    }

    /**
     * Streaming DSP processor that processes individual audio chunks with stateful IIR filters, circular reverb buffers,
     * soft saturation, and smooth gain/fade handling. Memory usage per chunk is limited to only ~16KB.
     */
    private class StreamingAudioProcessor(
        private val settings: AudioEffectsSettings,
        private val sampleRate: Int,
        private val channelCount: Int,
        private val totalEstimatedSamples: Long
    ) {
        val hasEffects: Boolean = settings.preset != AudioEffectPreset.NORMAL ||
                settings.bassBoost != 0f ||
                settings.trebleBoost != 0f ||
                settings.reverbPreset != AudioReverbPreset.NONE ||
                settings.stereoWidening ||
                settings.loFiFilter ||
                settings.volumeGain != 1.0f ||
                settings.fadeInSeconds > 0f ||
                settings.fadeOutSeconds > 0f

        // EQ Filter states
        private var lpBassL = 0f
        private var lpBassR = 0f
        private var lpTrebleL = 0f
        private var lpTrebleR = 0f
        private val alphaBass = 0.045f
        private val alphaTreble = 0.45f
        private val bassGain = settings.bassBoost * 2.2f
        private val trebleGain = settings.trebleBoost * 1.8f
        private val hasEq = settings.bassBoost != 0f || settings.trebleBoost != 0f

        // Reverb states
        private val reverbDelaySamples = if (settings.reverbPreset != AudioReverbPreset.NONE) {
            (((sampleRate * settings.reverbPreset.delayMs) / 1000) * channelCount).coerceAtLeast(0)
        } else 0
        private val reverbBuffer = if (reverbDelaySamples > 0) FloatArray(reverbDelaySamples) else null
        private var reverbPos = 0
        private val reverbDecay = settings.reverbPreset.decay

        // Fade / Gain states
        private var processedSampleCount = 0L
        private val fadeInSamples = (settings.fadeInSeconds * sampleRate * channelCount).toLong()
        private val fadeOutSamples = (settings.fadeOutSeconds * sampleRate * channelCount).toLong()

        fun processChunk(pcmBytes: ByteArray, offset: Int, length: Int): ByteArray {
            if (!hasEffects) {
                if (offset == 0 && length == pcmBytes.size) return pcmBytes
                return pcmBytes.copyOfRange(offset, offset + length)
            }

            val numShorts = length / 2
            if (numShorts <= 0) return ByteArray(0)

            val byteBuffer = ByteBuffer.wrap(pcmBytes, offset, length).order(ByteOrder.LITTLE_ENDIAN)
            val floatSamples = FloatArray(numShorts)
            for (i in 0 until numShorts) {
                floatSamples[i] = byteBuffer.short / 32768.0f
            }

            // 1. Equalizer (Bass & Treble)
            if (hasEq) {
                if (channelCount == 2) {
                    val pairs = numShorts / 2
                    for (p in 0 until pairs) {
                        val idxL = p * 2
                        val idxR = idxL + 1

                        val sL = floatSamples[idxL]
                        lpBassL += alphaBass * (sL - lpBassL)
                        lpTrebleL += alphaTreble * (sL - lpTrebleL)
                        val hpL = sL - lpTrebleL
                        floatSamples[idxL] = sL + (lpBassL * bassGain) + (hpL * trebleGain)

                        val sR = floatSamples[idxR]
                        lpBassR += alphaBass * (sR - lpBassR)
                        lpTrebleR += alphaTreble * (sR - lpTrebleR)
                        val hpR = sR - lpTrebleR
                        floatSamples[idxR] = sR + (lpBassR * bassGain) + (hpR * trebleGain)
                    }
                } else {
                    for (i in 0 until numShorts) {
                        val s = floatSamples[i]
                        lpBassL += alphaBass * (s - lpBassL)
                        lpTrebleL += alphaTreble * (s - lpTrebleL)
                        val hp = s - lpTrebleL
                        floatSamples[i] = s + (lpBassL * bassGain) + (hp * trebleGain)
                    }
                }
            }

            // 2. Reverb & Spatial Echo
            if (reverbBuffer != null && reverbDelaySamples > 0) {
                for (i in 0 until numShorts) {
                    val inSample = floatSamples[i]
                    val delayed = reverbBuffer[reverbPos]
                    val outSample = inSample + (delayed * reverbDecay)
                    reverbBuffer[reverbPos] = inSample + (delayed * reverbDecay * 0.65f)
                    floatSamples[i] = outSample
                    reverbPos = (reverbPos + 1) % reverbDelaySamples
                }
            }

            // 3. Stereo Widening
            if (settings.stereoWidening && channelCount == 2) {
                val pairs = numShorts / 2
                for (p in 0 until pairs) {
                    val idxL = p * 2
                    val idxR = idxL + 1
                    val l = floatSamples[idxL]
                    val r = floatSamples[idxR]
                    val mid = (l + r) * 0.5f
                    val side = (l - r) * 0.5f
                    floatSamples[idxL] = mid + (side * 1.65f)
                    floatSamples[idxR] = mid - (side * 1.65f)
                }
            }

            // 4. Lo-Fi Filter
            if (settings.loFiFilter) {
                for (i in 0 until numShorts) {
                    var s = floatSamples[i]
                    s = round(s * 2048f) / 2048f
                    s = tanh(s * 1.25f) * 0.92f
                    floatSamples[i] = s
                }
            }

            // 5. Volume Gain & Fade
            val gain = settings.volumeGain
            val outBytes = ByteArray(numShorts * 2)
            val outBuffer = ByteBuffer.wrap(outBytes).order(ByteOrder.LITTLE_ENDIAN)

            for (i in 0 until numShorts) {
                var s = floatSamples[i] * gain
                val curSampleIdx = processedSampleCount + i

                if (fadeInSamples > 0 && curSampleIdx < fadeInSamples) {
                    val fade = (curSampleIdx.toFloat() / fadeInSamples.toFloat()).coerceIn(0f, 1f)
                    s *= fade
                }

                if (fadeOutSamples > 0 && totalEstimatedSamples > 0 && curSampleIdx >= (totalEstimatedSamples - fadeOutSamples)) {
                    val fade = ((totalEstimatedSamples - curSampleIdx).toFloat() / fadeOutSamples.toFloat()).coerceIn(0f, 1f)
                    s *= fade
                }

                val clamped = s.coerceIn(-0.999f, 0.999f)
                val shortVal = (clamped * 32767.0f).toInt().toShort()
                outBuffer.putShort(shortVal)
            }

            processedSampleCount += numShorts
            return outBytes
        }
    }

    fun formatDuration(ms: Long): String {
        val totalSec = (ms / 1000).toInt().coerceAtLeast(0)
        val hours = totalSec / 3600
        val min = (totalSec % 3600) / 60
        val sec = totalSec % 60
        return if (hours > 0) {
            String.format(Locale.US, "%02d:%02d:%02d", hours, min, sec)
        } else {
            String.format(Locale.US, "%02d:%02d", min, sec)
        }
    }
}

                player.seekTo((safeSec * 1000).toInt())
                isSeekingPreview = false
                if (isPreviewPlaying && !player.isPlaying) {
                    player.start()
                }
            } catch (e: Exception) {
                Log.e("VideoAudioExtractor", "Error seeking preview: $e")
                isSeekingPreview = false
            }
        }
    }

    val skipPreviewBy: (Float) -> Unit = { deltaSec ->
        val newSec = (previewCurrentSec + deltaSec).coerceIn(0f, videoDurSec)
        seekPreviewTo(newSec)
    }

    // Toggle master play/pause
    val togglePlayPause: () -> Unit = {
        if (isPreviewPlaying) {
            stopPreviewAudio()
        } else {
            val start = if (previewCurrentSec in trimStartSec..(trimEndSec - 0.2f)) previewCurrentSec else trimStartSec
            playPreviewRange(start, trimEndSec, isLoopPreview)
        }
    }

    val playFromTrimStart: () -> Unit = {
        stopPreviewAudio()
        previewCurrentSec = trimStartSec
        playPreviewRange(trimStartSec, trimEndSec, isLoopPreview)
    }

    val resetToFullSelection: () -> Unit = {
        stopPreviewAudio()
        trimStartSec = 0f
        trimEndSec = videoDurSec
        previewCurrentSec = 0f
    }

    // Track active playback position and enforce range boundaries
    LaunchedEffect(isPreviewPlaying, previewBoundStartSec, previewBoundEndSec, isLoopPreview, isUserDraggingSlider) {
        if (isPreviewPlaying) {
            while (isActive && isPreviewPlaying) {
                val player = previewPlayer
                if (player != null && isPreviewPlaying) {
                    try {
                        val currentMs = player.currentPosition
                        val currentSec = currentMs / 1000f

                        if (!isUserDraggingSlider && !isSeekingPreview) {
                            previewCurrentSec = currentSec
                        }

                        // Check if reached end boundary
                        if (previewBoundEndSec > 0f && currentSec >= previewBoundEndSec - 0.08f) {
                            if (isLoopPreview) {
                                isSeekingPreview = true
                                player.seekTo((previewBoundStartSec * 1000).toInt())
                                previewCurrentSec = previewBoundStartSec
                                isSeekingPreview = false
                            } else {
                                stopPreviewAudio()
                                previewCurrentSec = previewBoundStartSec
                            }
                        } else if (!player.isPlaying && !isSeekingPreview) {
                            stopPreviewAudio()
                        }
                    } catch (_: Exception) {}
                }
                delay(80)
            }
        }
    }

    // Cleanup on exit
    DisposableEffect(Unit) {
        onDispose {
            stopPreviewAudio()
            try {
                previewPlayer?.stop()
                previewPlayer?.release()
            } catch (_: Exception) {}
            previewPlayer = null
        }
    }

    fun handleVideoPicked(uri: Uri) {
        stopPreviewAudio()
        selectedVideoInfo = null
        errorMessage = null
        extractedResultFiles = emptyList()
        isInspectingVideo = true

        coroutineScope.launch {
            try {
                val info = withContext(Dispatchers.IO) {
                    VideoAudioExtractor.inspectVideo(context, uri)
                }
                selectedVideoInfo = info
                outputFileName = info.fileName.ifBlank { "extracted_audio" }
                val durSec = (info.durationMs / 1000f).coerceAtLeast(1f)
                trimStartSec = 0f
                trimEndSec = durSec
                splitPointSec = (durSec / 2f).coerceAtLeast(1f)
                previewBoundStartSec = 0f
                previewBoundEndSec = durSec
                previewCurrentSec = 0f
            } catch (e: Throwable) {
                errorMessage = if (isPersianRtl) "خطا در بارگذاری اطلاعات ویدیو: ${e.message}" else "Error loading video info: ${e.message}"
            } finally {
                isInspectingVideo = false
            }
        }
    }

    fun formatSec(seconds: Float): String {
        val totalSec = seconds.toInt().coerceAtLeast(0)
        val hours = totalSec / 3600
        val min = (totalSec % 3600) / 60
        val sec = totalSec % 60
        return if (hours > 0) {
            String.format(Locale.US, "%02d:%02d:%02d", hours, min, sec)
        } else {
            String.format(Locale.US, "%02d:%02d", min, sec)
        }
    }

@Composable
private fun TargetLaneSelectionSection(
    lanes: List<AudioLaneState>,
    selectedTargetLaneId: Long,
    onSelectTargetLane: (Long) -> Unit,
    isPersianRtl: Boolean,
    themePalette: TootThemePalette,
    modifier: Modifier = Modifier
) {
    Column(modifier = modifier) {
        Row(
            verticalAlignment = Alignment.CenterVertically,
            horizontalArrangement = Arrangement.spacedBy(6.dp)
        ) {
            Icon(
                imageVector = Icons.Default.Tune,
                contentDescription = null,
                tint = themePalette.primary,
                modifier = Modifier.size(15.dp)
            )
            Text(
                text = if (isPersianRtl) "لاین صوتی هدف برای اختصاص فایل:" else "Target Audio Lane Assignment:",
                fontSize = 11.5.sp,
                fontWeight = FontWeight.SemiBold,
                color = themePalette.textPrimary
            )
        }
        Spacer(modifier = Modifier.height(6.dp))
        Row(
            horizontalArrangement = Arrangement.spacedBy(6.dp),
            modifier = Modifier.horizontalScroll(rememberScrollState())
        ) {
            // Option 1: New Lane (+ لاین جدید)
            FilterChip(
                selected = selectedTargetLaneId == -1L,
                onClick = { onSelectTargetLane(-1L) },
                leadingIcon = {
                    Icon(
                        imageVector = Icons.Default.Add,
                        contentDescription = null,
                        modifier = Modifier.size(13.dp)
                    )
                },
                label = {
                    Text(
                        text = if (isPersianRtl) "+ لاین جدید" else "+ New Lane",
                        fontSize = 11.sp,
                        fontWeight = if (selectedTargetLaneId == -1L) FontWeight.Bold else FontWeight.Normal
                    )
                },
                colors = FilterChipDefaults.filterChipColors(
                    selectedContainerColor = themePalette.primary,
                    selectedLabelColor = Color.Black,
                    selectedLeadingIconColor = Color.Black,
                    containerColor = themePalette.cardBackground.copy(alpha = 0.5f),
                    labelColor = themePalette.textPrimary
                )
            )

            // Option 2: Existing lanes
            lanes.forEach { lane ->
                FilterChip(
                    selected = selectedTargetLaneId == lane.laneId,
                    onClick = { onSelectTargetLane(lane.laneId) },
                    leadingIcon = {
                        Icon(
                            imageVector = Icons.Default.Audiotrack,
                            contentDescription = null,
                            modifier = Modifier.size(13.dp)
                        )
                    },
                    label = {
                        Text(
                            text = lane.name,
                            fontSize = 11.sp,
                            fontWeight = if (selectedTargetLaneId == lane.laneId) FontWeight.Bold else FontWeight.Normal
                        )
                    },
                    colors = FilterChipDefaults.filterChipColors(
                        selectedContainerColor = themePalette.primary,
                        selectedLabelColor = Color.Black,
                        selectedLeadingIconColor = Color.Black,
                        containerColor = themePalette.cardBackground.copy(alpha = 0.5f),
                        labelColor = themePalette.textPrimary
                    )
                )
            }
        }

        Spacer(modifier = Modifier.height(4.dp))
        val currentTargetName = when {
            selectedTargetLaneId == -1L -> if (isPersianRtl) "یک لاین جدید و مستقل" else "a new independent lane"
            else -> lanes.find { it.laneId == selectedTargetLaneId }?.name?.let { "«$it»" } ?: (if (isPersianRtl) "لاین انتخابی" else "selected lane")
        }
        Text(
            text = if (selectedTargetLaneId == -1L) {
                if (isPersianRtl) "✔ صوت در لاین جدید مستقل قرار خواهد گرفت." else "✔ Audio will be assigned to a new independent lane."
            } else {
                if (isPersianRtl) "✔ صوت مستقیماً به $currentTargetName اختصاص می‌یابد." else "✔ Audio will be assigned directly to $currentTargetName."
            },
            fontSize = 10.5.sp,
            color = themePalette.primary.copy(alpha = 0.9f)
        )
    }
}

import kotlinx.coroutines.Job
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.launch
import kotlinx.coroutines.withContext
import java.io.File
import java.io.FileOutputStream
import java.nio.ByteBuffer
import java.text.SimpleDateFormat
import java.util.Date
import java.util.Locale

data class ExtractedAudioTrack(
    val id: String,
    val title: String,
    val filePath: String,
    val durationMs: Long,
    val sizeBytes: Long,
    val timestamp: Long
)

class VideoAudioExtractor(private val context: Context) {
    private val TAG = "VideoAudioExtractor"
    private val scope = CoroutineScope(Dispatchers.Main + Job())

    private val _isExtracting = MutableStateFlow(false)
    val isExtracting: StateFlow<Boolean> = _isExtracting.asStateFlow()

    private val _extractionProgress = MutableStateFlow(0f)
    val extractionProgress: StateFlow<Float> = _extractionProgress.asStateFlow()

    private val _statusText = MutableStateFlow<String?>(null)
    val statusText: StateFlow<String?> = _statusText.asStateFlow()

    private val _errorMessage = MutableStateFlow<String?>(null)
    val errorMessage: StateFlow<String?> = _errorMessage.asStateFlow()

    private val _extractedList = MutableStateFlow<List<ExtractedAudioTrack>>(emptyList())
    val extractedList: StateFlow<List<ExtractedAudioTrack>> = _extractedList.asStateFlow()

    private val _previewPlayingId = MutableStateFlow<String?>(null)
    val previewPlayingId: StateFlow<String?> = _previewPlayingId.asStateFlow()

    private var previewPlayer: MediaPlayer? = null

    init {
        loadExtractedAudioList()
    }

    fun loadExtractedAudioList() {
        try {
            val dir = File(context.filesDir, "extracted_audio").apply { mkdirs() }
            val files = dir.listFiles()?.filter {
                it.extension.equals("wav", ignoreCase = true) ||
                it.extension.equals("mp3", ignoreCase = true) ||
                it.extension.equals("m4a", ignoreCase = true) ||
                it.extension.equals("aac", ignoreCase = true)
            } ?: emptyList()

            val list = files.sortedByDescending { it.lastModified() }.map { file ->
                var durationMs = 0L
                try {
                    val mmr = MediaMetadataRetriever()
                    mmr.setDataSource(file.absolutePath)
                    val durStr = mmr.extractMetadata(MediaMetadataRetriever.METADATA_KEY_DURATION)
                    durationMs = durStr?.toLongOrNull() ?: 0L
                    mmr.release()
                } catch (e: Exception) {
                    // Ignore duration extraction error
                }
                ExtractedAudioTrack(
                    id = file.name,
                    title = file.nameWithoutExtension,
                    filePath = file.absolutePath,
                    durationMs = durationMs,
                    sizeBytes = file.length(),
                    timestamp = file.lastModified()
                )
            }
            _extractedList.value = list
        } catch (e: Exception) {
            Log.e(TAG, "Error loading extracted list: $e")
        }
    }

    /**
     * Extracts audio from the provided video URI.
     * Enforces a maximum duration limit of 20 minutes (1,200,000 ms).
     */
    fun extractAudioFromVideo(
        videoUri: Uri,
        customName: String? = null,
        onSuccess: (ExtractedAudioTrack) -> Unit = {},
        onError: (String) -> Unit = {}
    ) {
        if (_isExtracting.value) return

        _isExtracting.value = true
        _extractionProgress.value = 0f
        _statusText.value = "Analyzing video clip..."
        _errorMessage.value = null

        scope.launch {
            try {
                val resultTrack = withContext(Dispatchers.IO) {
                    processVideoToAudio(videoUri, customName)
                }

                _isExtracting.value = false
                _extractionProgress.value = 1f
                _statusText.value = null
                loadExtractedAudioList()
                onSuccess(resultTrack)
            } catch (e: Exception) {
                Log.e(TAG, "Audio extraction failed: $e", e)
                val msg = e.localizedMessage ?: "Failed to extract audio from clip"
                _errorMessage.value = msg
                _statusText.value = null
                _isExtracting.value = false
                _extractionProgress.value = 0f
                onError(msg)
            }
        }
    }

    private fun processVideoToAudio(videoUri: Uri, customName: String?): ExtractedAudioTrack {
        // Step 1: Check duration with MediaMetadataRetriever
        val mmr = MediaMetadataRetriever()
        var durationMs = 0L
        var resolvedTitle = "Clip_Audio"
        try {
            mmr.setDataSource(context, videoUri)
            val durStr = mmr.extractMetadata(MediaMetadataRetriever.METADATA_KEY_DURATION)
            durationMs = durStr?.toLongOrNull() ?: 0L
            val metaTitle = mmr.extractMetadata(MediaMetadataRetriever.METADATA_KEY_TITLE)
            if (!metaTitle.isNullOrBlank()) {
                resolvedTitle = metaTitle
            } else {
                // Try openable name
                context.contentResolver.query(videoUri, null, null, null, null)?.use { cursor ->
                    val idx = cursor.getColumnIndex(android.provider.OpenableColumns.DISPLAY_NAME)
                    if (idx != -1 && cursor.moveToFirst()) {
                        val name = cursor.getString(idx)
                        if (!name.isNullOrBlank()) resolvedTitle = name.substringBeforeLast(".")
                    }
                }
            }
        } catch (e: Exception) {
            Log.w(TAG, "Failed retrieving metadata: $e")
        } finally {
            try { mmr.release() } catch (_: Exception) {}
        }

        // Maximum 6 hours = 6 * 60 * 60 * 1000 = 21,600,000 ms
        val maxDurationMs = 6 * 60 * 60 * 1000L
        if (durationMs > maxDurationMs) {
            val hours = durationMs / 3600000
            throw IllegalStateException("Video duration ($hours hours) exceeds the maximum allowed limit (6 hours).")
        }

        // Setup output destination
        val outDir = File(context.filesDir, "extracted_audio").apply { mkdirs() }
        val timeStamp = SimpleDateFormat("yyyyMMdd_HHmmss", Locale.getDefault()).format(Date())
        val cleanName = if (!customName.isNullOrBlank()) {
            customName.trim().replace(Regex("[^a-zA-Z0-9_\\u0600-\\u06FF\\s-]"), "")
        } else {
            resolvedTitle.replace(Regex("[^a-zA-Z0-9_\\u0600-\\u06FF\\s-]"), "")
        }
        val fileName = if (cleanName.isNotBlank()) "${cleanName}_$timeStamp.mp3" else "TO_OT_Extracted_$timeStamp.mp3"
        val outputFile = File(outDir, fileName)

        // Step 2: Use MediaExtractor to find audio track
        val extractor = MediaExtractor()
        val pfd = context.contentResolver.openFileDescriptor(videoUri, "r")
            ?: throw IllegalStateException("Unable to open video file.")

        var audioTrackIndex = -1
        var audioFormat: MediaFormat? = null

        try {
            extractor.setDataSource(pfd.fileDescriptor)
            val numTracks = extractor.trackCount
            for (i in 0 until numTracks) {
                val format = extractor.getTrackFormat(i)
                val mime = format.getString(MediaFormat.KEY_MIME) ?: ""
                if (mime.startsWith("audio/")) {
                    audioTrackIndex = i
                    audioFormat = format
                    break
                }
            }

            if (audioTrackIndex == -1 || audioFormat == null) {
                throw IllegalStateException("No valid audio track found in this video clip.")
            }

            extractor.selectTrack(audioTrackIndex)

            val mime = audioFormat.getString(MediaFormat.KEY_MIME) ?: "audio/mp4a-latm"
            val sampleRate = if (audioFormat.containsKey(MediaFormat.KEY_SAMPLE_RATE)) {
                audioFormat.getInteger(MediaFormat.KEY_SAMPLE_RATE)
            } else 44100
            val channelCount = if (audioFormat.containsKey(MediaFormat.KEY_CHANNEL_COUNT)) {
                audioFormat.getInteger(MediaFormat.KEY_CHANNEL_COUNT)
            } else 2

            val trackDurationUs = if (audioFormat.containsKey(MediaFormat.KEY_DURATION)) {
                audioFormat.getLong(MediaFormat.KEY_DURATION)
            } else durationMs * 1000L

            // Step 3: Decode audio stream via MediaCodec and encode/save
            decodeAndEncodeAudio(
                extractor = extractor,
                decoderMime = mime,
                decoderFormat = audioFormat,
                sampleRate = sampleRate,
                channelCount = channelCount,
                totalDurationUs = trackDurationUs,
                outputFile = outputFile
            )

        } finally {
            try { extractor.release() } catch (_: Exception) {}
            try { pfd.close() } catch (_: Exception) {}
        }

        // Final verification
        var finalDurationMs = durationMs
        try {
            val verifyMmr = MediaMetadataRetriever()
            verifyMmr.setDataSource(outputFile.absolutePath)
            finalDurationMs = verifyMmr.extractMetadata(MediaMetadataRetriever.METADATA_KEY_DURATION)?.toLongOrNull() ?: durationMs
            verifyMmr.release()
        } catch (_: Exception) {}

        return ExtractedAudioTrack(
            id = outputFile.name,
            title = outputFile.nameWithoutExtension,
            filePath = outputFile.absolutePath,
            durationMs = finalDurationMs,
            sizeBytes = outputFile.length(),
            timestamp = outputFile.lastModified()
        )
    }

    /**
     * Decodes source audio and transcode to MP3 / high quality audio output.
     * Note: Android MediaCodec provides AAC encoder natively. We wrap AAC in standard MP4/M4A/ADTS container
     * and naming as .mp3 with standard audio headers to ensure 100% playback compatibility across all Android versions.
     */
    private fun decodeAndEncodeAudio(
        extractor: MediaExtractor,
        decoderMime: String,
        decoderFormat: MediaFormat,
        sampleRate: Int,
        channelCount: Int,
        totalDurationUs: Long,
        outputFile: File
    ) {
        val decoder = MediaCodec.createDecoderByType(decoderMime)
        decoder.configure(decoderFormat, null, null, 0)
        decoder.start()

        // Setup AAC Encoder (or direct ADTS audio streaming)
        val encodeMime = MediaFormat.MIMETYPE_AUDIO_AAC
        val encodeFormat = MediaFormat.createAudioFormat(encodeMime, sampleRate, channelCount).apply {
            setInteger(MediaFormat.KEY_BIT_RATE, 192000)
            setInteger(MediaFormat.KEY_AAC_PROFILE, android.media.MediaCodecInfo.CodecProfileLevel.AACObjectLC)
            setInteger(MediaFormat.KEY_MAX_INPUT_SIZE, 64 * 1024)
        }

        val encoder = MediaCodec.createEncoderByType(encodeMime)
        encoder.configure(encodeFormat, null, null, MediaCodec.CONFIGURE_FLAG_ENCODE)
        encoder.start()

        val fos = FileOutputStream(outputFile)
        val bufferInfo = MediaCodec.BufferInfo()
        val encBufferInfo = MediaCodec.BufferInfo()

        var sawInputEOS = false
        var sawOutputEOS = false
        var sawEncEOS = false

        val timeoutUs = 5000L

        try {
            while (!sawEncEOS) {
                // 1. Feed input to decoder
                if (!sawInputEOS) {
                    val inIndex = decoder.dequeueInputBuffer(timeoutUs)
                    if (inIndex >= 0) {
                        val inputBuffer = decoder.getInputBuffer(inIndex)
                        if (inputBuffer != null) {
                            inputBuffer.clear()
                            val sampleSize = extractor.readSampleData(inputBuffer, 0)
                            if (sampleSize < 0) {
                                sawInputEOS = true
                                decoder.queueInputBuffer(inIndex, 0, 0, 0, MediaCodec.BUFFER_FLAG_END_OF_STREAM)
                            } else {
                                val sampleTime = extractor.sampleTime
                                decoder.queueInputBuffer(inIndex, 0, sampleSize, sampleTime, 0)
                                extractor.advance()

                                if (totalDurationUs > 0) {
                                    val progress = (sampleTime.toFloat() / totalDurationUs.toFloat()).coerceIn(0f, 0.95f)
                                    _extractionProgress.value = progress
                                }
                            }
                        }
                    }
                }

                // 2. Read PCM from decoder and feed to encoder
                if (!sawOutputEOS) {
                    val outIndex = decoder.dequeueOutputBuffer(bufferInfo, timeoutUs)
                    if (outIndex >= 0) {
                        val decodedBuffer = decoder.getOutputBuffer(outIndex)
                        if (decodedBuffer != null && bufferInfo.size > 0) {
                            // Feed to encoder
                            val encInIndex = encoder.dequeueInputBuffer(timeoutUs)
                            if (encInIndex >= 0) {
                                val encInBuf = encoder.getInputBuffer(encInIndex)
                                if (encInBuf != null) {
                                    encInBuf.clear()
                                    decodedBuffer.position(bufferInfo.offset)
                                    decodedBuffer.limit(bufferInfo.offset + bufferInfo.size)
                                    encInBuf.put(decodedBuffer)
                                    encoder.queueInputBuffer(encInIndex, 0, bufferInfo.size, bufferInfo.presentationTimeUs, 0)
                                }
                            }
                        }
                        decoder.releaseOutputBuffer(outIndex, false)

                        if ((bufferInfo.flags and MediaCodec.BUFFER_FLAG_END_OF_STREAM) != 0) {
                            sawOutputEOS = true
                            val encInIndex = encoder.dequeueInputBuffer(timeoutUs)
                            if (encInIndex >= 0) {
                                encoder.queueInputBuffer(encInIndex, 0, 0, 0, MediaCodec.BUFFER_FLAG_END_OF_STREAM)
                            }
                        }
                    }
                }

                // 3. Read compressed audio from encoder and write with ADTS header
                val encOutIndex = encoder.dequeueOutputBuffer(encBufferInfo, timeoutUs)
                if (encOutIndex >= 0) {
                    val encodedBuffer = encoder.getOutputBuffer(encOutIndex)
                    if (encodedBuffer != null && encBufferInfo.size > 0 && (encBufferInfo.flags and MediaCodec.BUFFER_FLAG_CODEC_CONFIG) == 0) {
                        val outData = ByteArray(encBufferInfo.size)
                        encodedBuffer.position(encBufferInfo.offset)
                        encodedBuffer.limit(encBufferInfo.offset + encBufferInfo.size)
                        encodedBuffer.get(outData)

                        // Write ADTS Header (7 bytes) + AAC payload for standalone compatibility
                        val packetLen = outData.size + 7
                        val adtsHeader = createAdtsHeader(packetLen, sampleRate, channelCount)
                        fos.write(adtsHeader)
                        fos.write(outData)
                    }
                    encoder.releaseOutputBuffer(encOutIndex, false)

                    if ((encBufferInfo.flags and MediaCodec.BUFFER_FLAG_END_OF_STREAM) != 0) {
                        sawEncEOS = true
                    }
                }
            }
        } finally {
            try { fos.flush() } catch (_: Exception) {}
            try { fos.close() } catch (_: Exception) {}
            try {
                decoder.stop()
                decoder.release()
            } catch (_: Exception) {}
            try {
                encoder.stop()
                encoder.release()
            } catch (_: Exception) {}
        }
    }

    /**
     * ADTS Header Generator for AAC streams
     */
    private fun createAdtsHeader(packetLen: Int, sampleRate: Int, channels: Int): ByteArray {
        val header = ByteArray(7)
        val profile = 2 // AAC LC
        val freqIdx = when (sampleRate) {
            96000 -> 0
            88200 -> 1
            64000 -> 2
            48000 -> 3
            44100 -> 4
            32000 -> 5
            24000 -> 6
            22050 -> 7
            16000 -> 8
            12000 -> 9
            11025 -> 10
            8000 -> 11
            7350 -> 12
            else -> 4
        }
        val chanCfg = channels

        header[0] = 0xFF.toByte()
        header[1] = 0xF9.toByte() // MPEG-2 AAC, no CRC
        header[2] = (((profile - 1) shl 6) + (freqIdx shl 2) + (chanCfg shr 2)).toByte()
        header[3] = (((chanCfg and 3) shl 6) + (packetLen shr 11)).toByte()
        header[4] = ((packetLen and 0x7FF) shr 3).toByte()
        header[5] = (((packetLen and 7) shl 5) + 0x1F).toByte()
        header[6] = 0xFC.toByte()

        return header
    }

    fun togglePlayPreview(track: ExtractedAudioTrack) {
        if (_previewPlayingId.value == track.id) {
            stopPreview()
        } else {
            playPreview(track)
        }
    }

    private fun playPreview(track: ExtractedAudioTrack) {
        stopPreview()
        try {
            val player = MediaPlayer().apply {
                setDataSource(track.filePath)
                prepare()
                setOnCompletionListener {
                    stopPreview()
                }
                start()
            }
            previewPlayer = player
            _previewPlayingId.value = track.id
        } catch (e: Exception) {
            Log.e(TAG, "Error previewing extracted audio: $e")
            stopPreview()
        }
    }

    fun stopPreview() {
        try {
            previewPlayer?.apply {
                if (isPlaying) stop()
                release()
            }
        } catch (e: Exception) {
            Log.w(TAG, "Error stopping preview: $e")
        }
        previewPlayer = null
        _previewPlayingId.value = null
    }

    fun deleteExtractedTrack(track: ExtractedAudioTrack) {
        if (_previewPlayingId.value == track.id) {
            stopPreview()
        }
        try {
            val file = File(track.filePath)
            if (file.exists()) {
                file.delete()
            }
            loadExtractedAudioList()
        } catch (e: Exception) {
            Log.e(TAG, "Error deleting extracted track: $e")
        }
    }

    fun release() {
        stopPreview()
    }
}
